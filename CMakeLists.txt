cmake_minimum_required(VERSION 3.20)
project(InterfaceConfigAPI VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(interface_config src/configuration.cpp)
add_library(interface_config::interface_config ALIAS interface_config)
target_include_directories(interface_config PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(interface_config PUBLIC cxx_std_20)
set_target_properties(interface_config PROPERTIES CXX_EXTENSIONS OFF)
add_executable(interface_config_example examples/main.cpp)
target_link_libraries(interface_config_example PRIVATE interface_config::interface_config)
set_target_properties(interface_config_example PROPERTIES CXX_EXTENSIONS OFF)
foreach(target IN ITEMS interface_config interface_config_example)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    endif()
endforeach()
